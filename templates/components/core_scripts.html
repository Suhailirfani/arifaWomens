// Darul Ma'arif Women's Academy - Core UI & Interaction Scripts function initCoreApp() { initTheme(); initMobileNav(); initAttendanceToggles(); } if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', initCoreApp); } else { initCoreApp(); } // Theme Management function initTheme() { var savedTheme = 'light'; try { savedTheme = localStorage.getItem('arifa_theme'); if (!savedTheme) { if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) { savedTheme = 'dark'; } else { savedTheme = 'light'; } } } catch (e) { savedTheme = 'light'; } document.documentElement.setAttribute('data-theme', savedTheme); if (typeof updateThemeIcon === 'function') { updateThemeIcon(savedTheme); } } function updateThemeIcon(theme) { var themeBtn = document.getElementById('theme-toggle-btn'); if (!themeBtn) return; if (theme === 'dark') { themeBtn.title = 'Switch to Light Mode'; themeBtn.setAttribute('aria-label', 'Switch to Light Mode'); themeBtn.innerHTML = ` `; } else { themeBtn.title = 'Switch to Dark Mode'; themeBtn.setAttribute('aria-label', 'Switch to Dark Mode'); themeBtn.innerHTML = ` `; } } // Mobile Menu Navigation function initMobileNav() { var toggleBtn = document.getElementById('mobile-menu-toggle'); var drawer = document.getElementById('mobile-drawer'); var backdrop = document.getElementById('mobile-backdrop'); var closeBtn = document.getElementById('mobile-close-btn'); if (toggleBtn && drawer && backdrop) { toggleBtn.addEventListener('click', () => { drawer.classList.add('open'); backdrop.classList.add('open'); }); var closeMenu = () => { drawer.classList.remove('open'); backdrop.classList.remove('open'); }; if (closeBtn) closeBtn.addEventListener('click', closeMenu); backdrop.addEventListener('click', closeMenu); } } // Attendance Sheet Interaction Helpers function initAttendanceToggles() { document.querySelectorAll('.attendance-toggle-btn').forEach(btn => { btn.addEventListener('click', function(e) { e.preventDefault(); var parent = this.closest('.attendance-toggle-group'); if (!parent) return; var studentId = parent.dataset.studentId; var targetStatus = this.dataset.status; var hiddenInput = document.getElementById('status_input_' + studentId); parent.querySelectorAll('.attendance-toggle-btn').forEach(b => { b.classList.remove('active-p', 'active-a', 'active-l'); }); if (targetStatus === 'PRESENT') this.classList.add('active-p'); if (targetStatus === 'ABSENT') this.classList.add('active-a'); if (targetStatus === 'LATE') this.classList.add('active-l'); if (hiddenInput) { hiddenInput.value = targetStatus; } updateAttendanceSummaryCounters(); }); }); var markAllPresentBtn = document.getElementById('btn-mark-all-present'); if (markAllPresentBtn) { markAllPresentBtn.addEventListener('click', () => { document.querySelectorAll('.attendance-toggle-group').forEach(group => { var studentId = group.dataset.studentId; var hiddenInput = document.getElementById('status_input_' + studentId); group.querySelectorAll('.attendance-toggle-btn').forEach(b => b.classList.remove('active-p', 'active-a', 'active-l')); var pBtn = group.querySelector('[data-status="PRESENT"]'); if (pBtn) pBtn.classList.add('active-p'); if (hiddenInput) hiddenInput.value = 'PRESENT'; }); updateAttendanceSummaryCounters(); }); } var markAllAbsentBtn = document.getElementById('btn-mark-all-absent'); if (markAllAbsentBtn) { markAllAbsentBtn.addEventListener('click', () => { document.querySelectorAll('.attendance-toggle-group').forEach(group => { var studentId = group.dataset.studentId; var hiddenInput = document.getElementById('status_input_' + studentId); group.querySelectorAll('.attendance-toggle-btn').forEach(b => b.classList.remove('active-p', 'active-a', 'active-l')); var aBtn = group.querySelector('[data-status="ABSENT"]'); if (aBtn) aBtn.classList.add('active-a'); if (hiddenInput) hiddenInput.value = 'ABSENT'; }); updateAttendanceSummaryCounters(); }); } } function updateAttendanceSummaryCounters() { var inputs = document.querySelectorAll('.attendance-status-hidden'); var p = 0, a = 0, l = 0; inputs.forEach(inp => { if (inp.value === 'PRESENT') p++; else if (inp.value === 'ABSENT') a++; else if (inp.value === 'LATE') l++; }); var countP = document.getElementById('count-present'); var countA = document.getElementById('count-absent'); var countL = document.getElementById('count-late'); if (countP) countP.textContent = p; if (countA) countA.textContent = a; if (countL) countL.textContent = l; }
